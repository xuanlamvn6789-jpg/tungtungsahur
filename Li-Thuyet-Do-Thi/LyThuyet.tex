\begin{frame}{2.3. Giả đồ thị (Pseudograph)}
    \centering
    \begin{tikzpicture}[every node/.style={circle,draw=mainblue,thick}]
        \node (1) {1}; \node (2) [right=of 1, xshift=1cm] {2};
        \draw (1) -- (2);

    \end{tikzpicture}
    \begin{detailbox}{mainblue}{Đặc điểm}

    \end{detailbox}

%--- SLIDE 4 ---
\begin{frame}{2.4. Đồ thị có hướng (Directed Graph)}
    \centering
    \begin{tikzpicture}[every node/.style={circle,draw=mainblue,thick}, >={Stealth}]
        \node (1) {1}; \node (2) [right=of 1, xshift=1cm] {2};
        \draw[->] (1) to [bend left] (2);
        \draw[->] (2) to [bend left] (1);
    \end{tikzpicture}
    \begin{itemize}
        \item Cạnh có hướng gọi là \textbf{cung}.
        \item Thứ tự cặp đỉnh $(u,v)$ là quan trọng.
    \end{itemize}
\end{frame}

%--- SLIDE 6 ---
\begin{frame}{2.6. Khái niệm Kề (Adjacent)}
    \centering
    \begin{tikzpicture}[every node/.style={circle,draw=mainblue,thick}]
        \node (1) {u}; \node (2) [right=of 1] {v}; \node (3) [below=of 2] {w};
        \draw (1) -- (2);
    \end{tikzpicture}
    \begin{itemize}
        \item $u$ và $v$ kề nhau vì có cạnh nối $(u,v)$.
        \item $v$ và $w$ không kề nhau.
    \end{itemize}
\end{frame}

%--- SLIDE 7 ---
\begin{frame}{2.7. Khái niệm Liên thuộc (Incident)}
    \centering
    \begin{tikzpicture}[every node/.style={circle,draw=mainblue,thick}]
        \node (1) {u}; \node (2) [right=of 1] {v};
        \draw (1) -- node[above, red] {$e$} (2);
    \end{tikzpicture}
    \begin{itemize}
        \item Cạnh $e$ liên thuộc với đỉnh $u$ và đỉnh $v$.
        \item Đỉnh $u, v$ là các \textbf{đầu mút} của cạnh $e$.
    \end{itemize}
\end{frame}

%--- SLIDE 8 ---
\begin{frame}{2.8. Bậc của đỉnh (Degree)}
    \begin{detailbox}{mainblue}{Định nghĩa}

    \end{detailbox}
    \centering
    \begin{tikzpicture}[every node/.style={circle,draw=mainblue,thick}]

        \draw (1)--(2) (1)--(3);
    \end{tikzpicture}
    \\ $d(1)=2, d(2)=1, d(3)=1$.
\end{frame}

\begin{frame}{2.9. Đỉnh treo và Đỉnh cô lập}
    \begin{itemize}
        \item \textbf{Đỉnh treo:} Bậc bằng 1.

%--- SLIDE 10 ---
\begin{frame}{2.10. Định lý Bắt tay}
    \begin{alertblock}{Định lý Handshaking}
        $\sum_{v \in V} d(v) = 2|E|$
    \end{alertblock}
    \vspace{0.3cm}
    \textit{Hệ quả: Tổng bậc của các đỉnh luôn là số chẵn. Do đó, số đỉnh có bậc lẻ phải là một số chẵn.}
\end{frame}

%--- SLIDE 11 ---
\begin{frame}{2.11. Bán bậc trong đồ thị có hướng}
    \begin{itemize}
        \item \textbf{Bán bậc vào ($d^-$):} Số cung đi vào đỉnh.
        \item \textbf{Bán bậc ra ($d^+$):} Số cung đi ra khỏi đỉnh.
    \end{itemize}
    \centering
    \begin{tikzpicture}[every node/.style={circle,draw=mainblue,thick}, >={Stealth}]
        \node (1) {v}; \draw[<-] (1) -- ++(1,1); \draw[->] (1) -- ++(1,-1);
    \end{tikzpicture}
    \\ $\sum d^-(v) = \sum d^+(v) = |E|$
\end{frame}

    \begin{columns}
        \begin{column}{0.5\textwidth}

        \end{column}
        \begin{column}{0.5\textwidth}
            \centering
            \begin{tikzpicture}[scale=0.7]
                \node[draw,circle](1){1}; \node[draw,circle,right=of 1](2){2};

            \end{tikzpicture}
        \end{column}
    \end{columns}
    \small $A[i][j]=1$ nếu đỉnh $i$ và $j$ kề nhau.
\end{frame}

\begin{frame}{2.25. Công thức Euler}
    \begin{alertblock}{Định lý Euler}
        $V - E + F = 2$
    \end{alertblock}
    \begin{itemize}
        \item $V$: Số đỉnh.
        \item $E$: Số cạnh.
        \item $F$: Số mặt (bao gồm cả mặt ngoài vô hạn).
    \end{itemize}
\end{frame}
