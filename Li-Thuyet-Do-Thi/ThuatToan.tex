\begin{frame}{Hôm nay nói gì?}
    \begin{block}{3 thuật toán trong đồ thị:}
        \begin{enumerate}
            \item \textbf{BFS}: Duyệt theo chiều rộng.
            \item \textbf{DFS}: Duyệt theo chiều sâu.
            \item \textbf{Dijkstra}: Tìm đường ngắn nhất.
        \end{enumerate}
    \end{block}
\end{frame}

% Slide 3
\begin{frame}{1. BFS - Duyệt theo chiều rộng}
    \begin{block}{�� Đặc điểm}
        \begin{itemize}
            \item Quét theo từng lớp.
            \item Gần $\rightarrow$ đi trước, xa $\rightarrow$ để sau.
            \item Xài \textbf{Queue} (Hàng đợi - xếp hàng ai đến trước phục vụ trước "FIFO").
        \end{itemize}
    \end{block}
    
    \begin{exampleblock}{�� Nhớ nhanh}
        \textbf{BFS = Lan ra như sóng.}
    \end{exampleblock}
\end{frame}

% Slide 5
\begin{frame}{BFS dùng để làm gì?}
    \begin{block}{Ứng dụng cực thực tế}
        \begin{itemize}
            \item Tìm đường đi ít bước nhất (đồ thị không trọng số).
            \item Gợi ý kết bạn chung (cách nhau 1-2 lớp bạn bè).
            \item Thuật toán loang (cái xô đổ màu trong Paint á).
        \end{itemize}
    \end{block}
\end{frame}

% Slide 6
\begin{frame}{2. DFS - Duyệt theo chiều sâu}
    \begin{block}{�� Đặc điểm}
        \begin{itemize}
            \item Cắm đầu đi 1 đường tới cùng.
            \item Bí đường / Hết đường $\rightarrow$ Quay xe lại bước trước(lấy lại đỉnh đã đi trước đó từ hàng đợi để xử lý tiếp các nhánh khác).
            \item Xài \textbf{Đệ quy} hoặc \textbf{Stack} (Ngăn xếp).
        \end{itemize}
    \end{block}
    
    \begin{exampleblock}{�� Nhớ nhanh}
        \textbf{DFS = Đi sâu rồi quay lại.}
    \end{exampleblock}
\end{frame}

% Slide 8
\begin{frame}{ BFS vs DFS khác nhau chổ nào?}
    \begin{block}{Khác nhau chỗ nào?}
        \begin{itemize}
            \item \textbf{Cách đi:} BFS dàn hàng ngang, DFS chọc thẳng 1 nhánh.
            \item \textbf{Lưu trữ:} BFS xài Queue (khá tốn RAM), DFS xài Stack/Đệ quy (nhẹ máy hơn).
        \end{itemize}
    \end{block}
\end{frame}

% Slide 9
\begin{frame}{Nên dùng cái nào?}
    \begin{exampleblock}{Tùy tình hình:}
        \begin{itemize}
            \item Cần tìm đường nhanh nhất, đích ở gần $\rightarrow$ Dùng \textbf{BFS}.
            \item Cây to đùng, sợ tốn bộ nhớ $\rightarrow$ Chơi \textbf{DFS}.
            \item Cứ bắt duyệt hết mọi đỉnh $\rightarrow$ Xài cái nào quen tay thì viết, thường DFS lẹ hơn.
        \end{itemize}
    \end{exampleblock}
\end{frame}

% Slide 12
\begin{frame}{Sao không dùng BFS cho lẹ mà phải Dijkstra?}
    \begin{block}{Tại vì thực tế nó khác}
        \begin{itemize}
            \item BFS coi đường nào cũng như nhau (đi 1 bước).
            \item Nhưng thực tế đường có đường 5km, có đường 20km. 
            \item BFS thấy ít bước hơn là đi, dễ đâm đầu vào cái đường 20km $\rightarrow$ \textbf{Toang!}
            \item Nên mới cần Dijkstra để tính tổng khoảng cách cho chuẩn.
        \end{itemize}
    \end{block}
\end{frame}
