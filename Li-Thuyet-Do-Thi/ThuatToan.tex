\section{BFS - Tìm kiếm theo chiều rộng}

% --- Slide 4: Khái niệm BFS ---
\begin{frame}{1. Thuật toán BFS (Tìm kiếm theo chiều rộng)}
    \begin{block}{Ý tưởng chính}
        Duyệt đồ thị theo từng "lớp" (layer). Bắt đầu từ đỉnh gốc, thăm tất cả các đỉnh kề nó, sau đó mới tiến xuống lớp tiếp theo.
    \end{block}
    \vspace{0.3cm}
    \begin{itemize}
        \item \textbf{Cấu trúc dữ liệu sử dụng:} Hàng đợi (Queue - FIFO).
        \item \textbf{Đặc điểm:} Đảm bảo tìm được đường đi ngắn nhất trên đồ thị không có trọng số.
        \item \textbf{Quy trình:}
        \begin{enumerate}
            \item Đưa đỉnh gốc vào Queue.
            \item Lấy đỉnh ở đầu Queue ra thăm.
            \item Đưa tất cả các đỉnh kề chưa thăm của nó vào cuối Queue.
            \item Lặp lại bước 2 và 3 cho đến khi Queue rỗng.
        \end{enumerate}
    \end{itemize}
\end{frame}

% --- Slide 5: BFS Step-by-step ---
\begin{frame}{BFS: Minh họa Step-by-step}
    \textbf{Bắt đầu từ đỉnh A.} Queue hiện tại: \only<1>{[A]}\only<2>{[B, C]}\only<3>{[C, D, E]}\only<4>{[D, E, F]}\only<5>{[E, F]}\only<6>{[F]}\only<7>{Rỗng}
    \vspace{0.3cm}
    
    \begin{center}
    \begin{tikzpicture}[
        node distance=2.2cm,
        main node/.style={circle, draw, font=\bfseries, minimum size=8mm},
        visited/.style={circle, draw, fill=OrangeAccent!60, font=\bfseries, minimum size=8mm},
        queue/.style={circle, draw, fill=DarkTeal!30, font=\bfseries, minimum size=8mm}
    ]
        % Khai báo các node và trạng thái theo từng step
        \node[main node]<1>[label=above:Start] (A) {A};
        \node[visited]<2->[label=above:Start] (A) {A};
        
        \node[main node]<1> (B) [below left of=A] {B};
        \node[queue]<2> (B) [below left of=A] {B};
        \node[visited]<3-> (B) [below left of=A] {B};
        
        \node[main node]<1-2> (C) [below right of=A] {C};
        \node[queue]<3> (C) [below right of=A] {C};
        \node[visited]<4-> (C) [below right of=A] {C};
        
        \node[main node]<1-2> (D) [below left of=B] {D};
        \node[queue]<3> (D) [below left of=B] {D};
        \node[visited]<4-> (D) [below left of=B] {D};

        \node[main node]<1-2> (E) [below right of=B] {E};
        \node[queue]<3-4> (E) [below right of=B] {E};
        \node[visited]<5-> (E) [below right of=B] {E};

        \node[main node]<1-3> (F) [below of=C] {F};
        \node[queue]<4-5> (F) [below of=C] {F};
        \node[visited]<6-> (F) [below of=C] {F};

        % Khai báo các cạnh
        \path[draw, thick]
        (A) edge (B) (A) edge (C)
        (B) edge (D) (B) edge (E)
        (C) edge (F);
    \end{tikzpicture}
    \end{center}
    
    \begin{itemize}
        \item<1-> \textbf{Bước 1:} Thăm A. Kề A là B, C. Đưa B, C vào Queue.
        \item<2-> \textbf{Bước 2:} Lấy B. Thăm B. Đưa D, E (kề B) vào Queue.
        \item<3-> \textbf{Bước 3:} Lấy C. Thăm C. Đưa F (kề C) vào Queue.
        \item<4-> \textbf{Bước 4,5,6:} Lần lượt lấy D, E, F ra khỏi Queue và thăm. Đồ thị duyệt xong!
    \end{itemize}
\end{frame}

% --- Slide 6: Ứng dụng BFS ---
\begin{frame}{Ứng dụng của thuật toán BFS}
    \begin{itemize}
        \item \textbf{Tìm đường đi ngắn nhất:} Trên đồ thị không trọng số (ví dụ: giải mê cung, số bước di chuyển ít nhất của quân mã trên bàn cờ vua).
        \item \textbf{Mạng xã hội:} Tìm những người bạn chung, gợi ý kết bạn (bạn bè bậc 1, bậc 2,...).
        \item \textbf{Hệ thống mạng (Broadcasting):} Gửi gói tin (packet) đến tất cả các node trong mạng.
        \item \textbf{Web Crawler:} Các công cụ tìm kiếm (như Google) dùng thuật toán tương tự BFS để thu thập các trang web theo từng cấp độ liên kết.
    \end{itemize}
\end{frame}

\section{DFS - Tìm kiếm theo chiều sâu}

% --- Slide 7: Khái niệm DFS ---
\begin{frame}{2. Thuật toán DFS (Tìm kiếm theo chiều sâu)}
    \begin{block}{Ý tưởng chính}
        Khám phá các nhánh sâu nhất có thể trước khi quay lui (backtracking) để khám phá các nhánh khác.
    \end{block}
    \vspace{0.3cm}
    \begin{itemize}
        \item \textbf{Cấu trúc dữ liệu sử dụng:} Ngăn xếp (Stack - LIFO) hoặc sử dụng Đệ quy (Recursion).
        \item \textbf{Quy trình cơ bản:}
        \begin{enumerate}
            \item Thăm một đỉnh kề chưa được thăm, đánh dấu đã thăm.
            \item Tiếp tục đi sâu vào đỉnh kề của đỉnh vừa thăm.
            \item Nếu gặp "ngõ cụt" (không còn đỉnh kề nào chưa thăm), quay lui (backtrack) về đỉnh trước đó.
            \item Lặp lại đến khi duyệt hết mọi đỉnh.
        \end{enumerate}
    \end{itemize}
\end{frame}

% --- Slide 8: DFS Step-by-step ---
\begin{frame}{DFS: Minh họa Step-by-step}
    \textbf{Bắt đầu từ đỉnh A.} Trạng thái duyệt: Đi sâu rồi quay lui.
    \vspace{0.3cm}
    
    \begin{center}
    \begin{tikzpicture}[
        node distance=2.2cm,
        main node/.style={circle, draw, font=\bfseries, minimum size=8mm},
        visited/.style={circle, draw, fill=DarkTeal!60, text=white, font=\bfseries, minimum size=8mm},
        active/.style={circle, draw, fill=OrangeAccent!60, font=\bfseries, minimum size=8mm}
    ]
        \node[main node]<1>[label=above:Start] (A) {A};
        \node[active]<2> (A) {A}; \node[visited]<3-> (A) {A};
        
        \node[main node]<1-2> (B) [below left of=A] {B};
        \node[active]<3> (B) {B}; \node[visited]<4-> (B) {B};
        
        \node[main node]<1-7> (C) [below right of=A] {C};
        \node[active]<8> (C) {C}; \node[visited]<9-> (C) {C};
        
        \node[main node]<1-3> (D) [below left of=B] {D};
        \node[active]<4> (D) {D}; \node[visited]<5-> (D) {D};

        \node[main node]<1-5> (E) [below right of=B] {E};
        \node[active]<6> (E) {E}; \node[visited]<7-> (E) {E};

        \node[main node]<1-8> (F) [below of=C] {F};
        \node[active]<9> (F) {F}; \node[visited]<10-> (F) {F};

        \path[draw, thick]
        (A) edge (B) (A) edge (C)
        (B) edge (D) (B) edge (E)
        (C) edge (F);
    \end{tikzpicture}
    \end{center}
    
    \begin{itemize}
        \item<1-> \textbf{Đường đi:} A $\rightarrow$ B $\rightarrow$ D $\rightarrow$ (Hết đường, quay lui B) $\rightarrow$ E $\rightarrow$ (Quay lui B) $\rightarrow$ (Quay lui A) $\rightarrow$ C $\rightarrow$ F.
        \item<10-> \textbf{Thứ tự duyệt xong:} A, B, D, E, C, F.
    \end{itemize}
\end{frame}

% --- Slide 9: So sánh BFS và DFS ---
\begin{frame}{So sánh BFS và DFS}
    \begin{table}[]
        \centering
        \renewcommand{\arraystretch}{1.5}
        \begin{tabular}{|p{3cm}|p{4cm}|p{4cm}|}
        \hline
        \rowcolor{DarkTeal}
        \textcolor{white}{\textbf{Tiêu chí}} & \textcolor{white}{\textbf{BFS}} & \textcolor{white}{\textbf{DFS}} \\ \hline
        \textbf{Cấu trúc dữ liệu} & Queue (Hàng đợi) & Stack (Ngăn xếp) / Đệ quy \\ \hline
        \textbf{Chiến lược} & Duyệt theo chiều ngang (Lớp) & Duyệt theo chiều sâu (Nhánh) \\ \hline
        \textbf{Ưu điểm} & Tìm được đường đi ngắn nhất (đồ thị không trọng số) & Tốn ít bộ nhớ hơn, code đệ quy ngắn gọn \\ \hline
        \textbf{Nhược điểm} & Tốn nhiều bộ nhớ để lưu trữ Queue ở các tầng rộng & Không đảm bảo đường đi tìm được là ngắn nhất \\ \hline
        \end{tabular}
    \end{table}
\end{frame}

\section{Dijkstra - Đường đi ngắn nhất}

% --- Slide 10: Khái niệm Dijkstra ---
\begin{frame}{3. Thuật toán Dijkstra}
    \begin{block}{Mục đích}
        Tìm đường đi ngắn nhất từ một đỉnh nguồn (source) đến tất cả các đỉnh còn lại trong đồ thị. \textbf{Điều kiện:} Trọng số các cạnh phải không âm ($\ge 0$).
    \end{block}
    \vspace{0.3cm}
    \textbf{Ý tưởng Step-by-step:}
    \begin{itemize}
        \item \textbf{Khởi tạo:} Khoảng cách từ điểm đầu đến chính nó là $0$, đến các điểm khác là $\infty$.
        \item \textbf{Cập nhật (Relaxation):} Lặp lại các bước sau:
        \begin{enumerate}
            \item Chọn đỉnh $u$ có khoảng cách nhỏ nhất chưa được cố định.
            \item Xét các đỉnh kề $v$ của $u$: Nếu khoảng cách hiện tại đến $v$ lớn hơn khoảng cách đi qua $u$, ta tiến hành cập nhật lại đường đi ngắn hơn.
            \item Đánh dấu $u$ đã cố định xong khoảng cách.
        \end{enumerate}
    \end{itemize}
\end{frame}

% --- Slide 11: Dijkstra Step 0 ---
\begin{frame}{Dijkstra: Đồ thị minh họa (Bước 0 - Khởi tạo)}
    \textbf{Yêu cầu:} Tìm đường đi ngắn nhất từ \textbf{A} đến các đỉnh.
    \vspace{0.3cm}
    
    \begin{center}
    \begin{tikzpicture}[
        node distance=3cm,
        main node/.style={circle, draw, font=\bfseries, minimum size=10mm},
        edge label/.style={fill=white, inner sep=1pt, font=\small}
    ]
        \node[main node, fill=OrangeAccent!30] (A) {A ($0$)};
        \node[main node] (B) [above right of=A] {B ($\infty$)};
        \node[main node] (C) [below right of=A] {C ($\infty$)};
        \node[main node] (D) [below right of=B] {D ($\infty$)};

        \path[draw, thick, ->] (A) edge node[edge label] {4} (B);
        \path[draw, thick, ->] (A) edge node[edge label] {1} (C);
        \path[draw, thick, ->] (B) edge node[edge label] {1} (D);
        \path[draw, thick, ->] (C) edge node[edge label] {2} (B);
        \path[draw, thick, ->] (C) edge node[edge label] {5} (D);
    \end{tikzpicture}
    \end{center}
    
    \begin{itemize}
        \item Khởi tạo khoảng cách: $dist[A] = 0$. Các đỉnh khác là $\infty$.
        \item Tập đỉnh đã cố định: \textbf{Rỗng}.
    \end{itemize}
\end{frame}

% --- Slide 12: Dijkstra Step 1 ---
\begin{frame}{Dijkstra: Bước 1 (Chọn đỉnh A)}
    \begin{center}
    \begin{tikzpicture}[
        node distance=3cm,
        main node/.style={circle, draw, font=\bfseries, minimum size=10mm},

        edge label/.style={fill=white, inner sep=1pt, font=\small}
    ]
        \node[fixed node] (A) {A ($0$)};
        \node[main node, fill=yellow!40] (B) [above right of=A] {B ($4$)};
        \node[main node, fill=yellow!40] (C) [below right of=A] {C ($1$)};
        \node[main node] (D) [below right of=B] {D ($\infty$)};

        \path[draw, thick, ->, OrangeAccent] (A) edge node[edge label] {4} (B);
        \path[draw, thick, ->, OrangeAccent] (A) edge node[edge label] {1} (C);
        \path[draw, thick, ->] (B) edge node[edge label] {1} (D);
        \path[draw, thick, ->] (C) edge node[edge label] {2} (B);
        \path[draw, thick, ->] (C) edge node[edge label] {5} (D);
    \end{tikzpicture}
    \end{center}
    
    \begin{itemize}
        \item Chọn \textbf{A} (có $dist=0$ nhỏ nhất). Cố định A.
        \item Cập nhật kề của A:
        \begin{itemize}
            \item $dist[B] = \min(\infty, 0 + 4) = 4$.
            \item $dist[C] = \min(\infty, 0 + 1) = 1$.
        \end{itemize}
    \end{itemize}
\end{frame}

    \end{itemize}

        \item \textbf{Kết quả khoảng cách ngắn nhất từ A:} B=3, C=1, D=4.
